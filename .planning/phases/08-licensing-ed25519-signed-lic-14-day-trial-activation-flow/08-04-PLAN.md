    Per CONTEXT D-13 verbatim:

    ```javascript
    #!/usr/bin/env node
    // Vendor-side license signing CLI (NOT bundled with the shipped app).
    // Usage: LICENSE_SIGNING_KEY_PATH=./secrets/ed25519.private node scripts/gen-license.cjs <machine-fingerprint> <vendor-id>
    // Reads the Ed25519 private key from the LICENSE_SIGNING_KEY_PATH env var (default ./secrets/ed25519.private),
    // produces a sidecar zip of license.json + license.sig, writes <machineId>.lic to cwd.

    const archiver = require('archiver');
    const ed = require('@noble/ed25519');
    const { sha512 } = require('@noble/hashes/sha2.js');
    const fs = require('node:fs');
    const path = require('node:path');

    ed.hashes.sha512 = sha512;

    async function main() {
      const [machineFingerprint, vendorId] = process.argv.slice(2);
      if (!machineFingerprint || !vendorId) {
        console.error('Usage: node scripts/gen-license.cjs <machine-fingerprint> <vendor-id>');
        console.error('  Set LICENSE_SIGNING_KEY_PATH to the private key file (default: ./secrets/ed25519.private)');
        process.exit(1);
      }

      const keyPath = process.env.LICENSE_SIGNING_KEY_PATH || path.join(process.cwd(), 'secrets', 'ed25519.private');
      let privateKey;
      try {
        const keyHex = fs.readFileSync(keyPath, 'utf8').trim();
        privateKey = Uint8Array.from(keyHex.split(',').map((s) => parseInt(s.trim(), 10)));
      } catch (err) {
        console.error(`Failed to read private key from ${keyPath}: ${err.message}`);
        console.error('Generate a key with: node -e "import(\'@noble/ed25519\').then(m => { const k = m.keygen(); console.log(Array.from(k.secretKey).join(\',\')); })"');
        process.exit(2);
      }

      if (privateKey.length !== 32) {
        console.error(`Private key must be exactly 32 bytes (got ${privateKey.length}). Comma-separated bytes expected.`);
        process.exit(2);
      }

      const payload = {
        vendorId,
        machineFingerprint,
        licensedAt: Date.now(),
        expiresAt: null, // perpetual
      };
      const json = Buffer.from(JSON.stringify(payload), 'utf8');
      const signature = Buffer.from(await ed.signAsync(json, privateKey));

      const outPath = `${machineFingerprint.slice(0, 12)}.lic`;
      const output = fs.createWriteStream(outPath);
      const archive = archiver('zip', { zlib: { level: 9 } });
      archive.pipe(output);
      archive.append(json, { name: 'license.json' });
      archive.append(signature, { name: 'license.sig' });
      await archive.finalize();

      // Wait for the output stream to finish before exiting.
      await new Promise((resolve, reject) => {
        output.on('close', resolve);
        output.on('error', reject);
      });

      console.log(`Wrote ${outPath}`);
    }

    main().catch((err) => {
      console.error(err);
      process.exit(1);
    });
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Renderer ↔ `LICENSE_ACTIVATE` IPC | Renderer supplies the `.lic` path from `dialog.showOpenDialog`; main reads + verifies |
| `loadAndVerifyLicense` ↔ file system | Reads the `.lic` zip; writes verified sidecar to `licenseDir()` |
| Vendor machine ↔ shipped binary | Vendor's private key is NEVER shipped; only the public key constant from `verify.ts` ships |
| `.gitignore` ↔ accidental commit | The catch-all in `secrets/.gitignore` is the last line of defense |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-08-L01 | Tampering | Tampered `.lic` (edited JSON or signature) | critical | mitigate | `loadAndVerifyLicense` runs `verifyLicense` on the raw bytes; any byte change → `{valid: false, reason: 'SIGNATURE_MISMATCH'}` → `IPC_LICENSE_INVALID` + `license.invalid` audit row; the sidecar is NEVER written on failure |
| T-08-L02 | Spoofing | License theft across machines | high | mitigate | `verifyLicense` compares the embedded `machineFingerprint` to the COMPUTED `computeMachineFingerprint()`; mismatch → `FINGERPRINT_MISMATCH` → `IPC_LICENSE_INVALID` |
| T-08-L03 | Spoofing | Public key replacement (attacker swaps the embedded key) | high | mitigate | `VENDOR_PUBLIC_KEY_HEX` is the embedded constant from `verify.ts`; the test asserts the embedded constant is used (via the roundtrip test in Plan 01). Plan 06 grep gate catches drift. |
| T-08-L04 | Elevation of Privilege | Vendor private key leaks into shipped binary | critical | mitigate | `.gitignore` excludes `secrets/` + `*.private` + `*.lic`; `secrets/.gitignore` is a catch-all; the `gen-license.cjs` script reads the key from `LICENSE_SIGNING_KEY_PATH` (env var only, NOT a hardcoded path); the shipped binary does NOT import or read the private key |
| T-08-L05 | Repudiation | License activation with no audit row | medium | mitigate | Every `loadAndVerifyLicense` invocation emits EITHER `license.activated` (success) OR `license.invalid` (failure) audit row with `userId: null` (workstation-level event per CONTEXT D-11) + `metadata: { vendorId?, fingerprintHash, reason?, expiresAt? }` |

| T-08-L07 | Information Disclosure | Sidecar files contain PII (machine fingerprint hash, vendor id, license timestamp) | low | accept | The sidecar files contain NO patient data, NO clinical data; only license metadata. The fingerprint is a SHA-256 hash (NOT a raw MAC/serial). The `audit_log` already records the same metadata (Plan 02 + this plan). Per Phase 7 D-13 + RESEARCH §Patterns: the renderer NEVER composes paths directly — `dialog.showOpenDialog` returns the only legitimate path the renderer sends. |
| T-08-SC | Tampering | npm install of `@noble/ed25519` / `@noble/hashes` | high | mitigate | Already verified in Plan 01's package legitimacy audit. `archiver` is already installed Phase 7. No new packages install in this plan. |

## Mitigations implemented by this plan

- **T-08-L01** (tampered license): `verifyLicense` returns `{valid: false, reason}` on any byte change; the sidecar is NEVER written on failure; audit row emits.
- **T-08-L02** (license theft): `verifyLicense` compares fingerprints; mismatch → `FINGERPRINT_MISMATCH` rejection.
- **T-08-L04** (private key leak): `.gitignore` + `secrets/.gitignore` catch-all; `LICENSE_SIGNING_KEY_PATH` env var; no hardcoded paths.
- **T-08-L05** (audit repudiation): every activation (success OR failure) writes an audit row with the correct metadata.
- **T-08-L08** (path injection): zod validation + yauzl-fails-on-non-zip + Phase 7 D-13 dialog-only path surface.

## Mitigations deferred to later plans

- **T-08-L06** (rate limiting): v1.1 follow-up. Not part of the v1 ship gate per REQUIREMENTS.
- **T-08-GREP** (drift in IPC gate coverage of `LICENSE_ACTIVATE`): Plan 06 grep gate verifies `LICENSE_ACTIVATE` is in EXEMPT_CHANNELS + is wrapped with `licenseGated` (for grep consistency, even though the gate is a no-op for exempt channels).
