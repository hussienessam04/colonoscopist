  trial: 'bg-blue-500',

  unactivated: 'bg-slate-500',
}[licenseState];

return (
  <Button ...>
    <KeyRound className="mr-2 size-4" />
    <span>{t('license.sidebarEntry')}</span>
    {status && (
      <>
        <span className={cn('ml-auto inline-block size-2 rounded-full', statusDotColor)} aria-hidden="true" />
        <span className="ml-1 text-xs text-muted-foreground">
          {licenseState === 'trial' && status.trialDaysRemaining !== undefined
            ? t(badgeKey as any, { count: status.trialDaysRemaining })
            : t(badgeKey as any)}
        </span>
      </>
    )}
  </Button>
);
