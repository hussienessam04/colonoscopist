  <options>
    <option id="option-a">

      <cons>If SET-04 grows role granularity (e.g. "viewer-only staff account") this requires a migration.</cons>
    </option>

      <name>Add a `role` column now (TEXT NULL) for forward compatibility</name>
      <pros>No migration later if roles are wanted.</pros>

    </option>
  </options>
