    Create `tests/integration/backup-restore-roundtrip.test.ts` (RUN_SMOKE=1 gated): D-16 verbatim roundtrip.
    ```ts
    import { describe, it, expect, beforeAll } from 'vitest';
    import { mkdtempSync, rmSync, existsSync, statSync, readdirSync } from 'node:fs';
    import { tmpdir } from 'node:os';
    import path from 'node:path';
    import { getDb } from '@/main/db';
    import { createBackup } from '@/main/backup';
    import { previewRestore, unpackRestore, integrityCheck } from '@/main/backup/restore';

    const runSmoke = !!process.env.RUN_SMOKE;
    const itSmoke = runSmoke ? it : it.skip;

    describe('backup → restore roundtrip', () => {
      let workDir: string;
      let stagingDir: string;

      beforeAll(() => {
        workDir = mkdtempSync(path.join(tmpdir(), 'phase7-roundtrip-'));
        stagingDir = path.join(workDir, 'staging');
      });

      itSmoke('creates backup + unpacks + integrity check passes', async () => {
        // Seed minimal data so the DB has content to backup.
        getDb().exec(`INSERT INTO users (id, full_name, is_first_admin, pin_hash, failed_attempts, is_locked, created_at)
                       VALUES ('u1', 'Test', 1, 'hash', 0, 0, ${Date.now()})`);
        const zipPath = path.join(workDir, 'backup.zip');
        const backupResult = await createBackup({ destZipPath: zipPath });
        expect(existsSync(zipPath)).toBe(true);
        expect(backupResult.sizeBytes).toBeGreaterThan(0);

        const preview = await previewRestore({ zipPath, stagingDir });
        expect(preview.dbIntegrityCheck).toBe('ok');

        const unpack = await unpackRestore({ zipPath, stagingDir });
        expect(unpack.fileCount).toBeGreaterThan(0);
        expect(existsSync(path.join(stagingDir, 'app.db'))).toBe(true);

        const finalCheck = integrityCheck(stagingDir);
        expect(finalCheck).toBe('ok');
      });

      afterAll(() => {
        if (existsSync(workDir)) rmSync(workDir, { recursive: true, force: true });
      });
    });
    ```

    ```ts
    import { describe, it, expect, beforeAll } from 'vitest';
    import { existsSync, statSync, readFileSync } from 'node:fs';
    import { getDb } from '@/main/db';
    import { reportsRepo } from '@/main/db/reports-repo';
    import { patientRepo } from '@/main/db/patients';
    import { proceduresRepo } from '@/main/db/procedures-repo';
    import { doctorProfileRepo } from '@/main/db/doctor-profile-repo';
    import { renderReportPdf } from '@/main/pdf/render-report-pdf';

    const runSmoke = !!process.env.RUN_SMOKE;
    const itSmoke = runSmoke ? it : it.skip;

    describe('AR PDF ship-gate (D-27)', () => {
      itSmoke('renders AR PDF with file > 50KB + PDF magic bytes', async () => {
        // Seed: user + doctor profile with language='ar' + patient + procedure + finalized report.
        getDb().exec(`INSERT INTO users (id, full_name, is_first_admin, pin_hash, failed_attempts, is_locked, created_at)
                       VALUES ('u1', 'د. أحمد', 1, 'hash', 0, 0, ${Date.now()})`);
        doctorProfileRepo.upsert({
          userId: 'u1',
          fullNameEn: 'Dr. Ahmed',
          fullNameAr: 'د. أحمد',
          clinicNameEn: 'Clinic',
          clinicNameAr: 'العيادة',
          address: null,
          phone: null,
          language: 'ar',
        });
        const patient = patientRepo.create({
          fullName: 'محمد علي',
          dob: '1980-04-12',
          gender: 'male',
          mrn: '12345',
          phone: null,
          notes: null,
        });
        const procedure = proceduresRepo.insert({
          patientId: patient.id,
          doctorId: 'u1',
          videoPath: 'video.mp4',
          presetSummary: { kind: 'sd', resolution: '720x480', framerate: 30, bitrate: '4M' },
          audioDeviceName: null,
        });
        proceduresRepo.updateFinalized(procedure.id, {
          endedAt: Date.now(),
          durationSeconds: 600,
          status: 'completed',
          videoPath: 'video.mp4',
        });
        const report = reportsRepo.create({
          procedureId: procedure.id,
          doctorId: 'u1',
        });

        const result = await renderReportPdf(report.id, { language: 'ar' });
        const stat = statSync(result.pdfPath);
        expect(stat.size).toBeGreaterThan(50_000);
        const buffer = readFileSync(result.pdfPath);
        expect(buffer.slice(0, 4).toString()).toBe('%PDF');
      }, 60_000); // 60s timeout for AR PDF render (Font.register + render)
    });
    ```
